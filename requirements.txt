# ─────────────────────────────────────────────────────────────────────────────
# POST-INSTALL PATCH (run once after pip install):
#
# thinc's xp2torch() checks for `toDlpack` before `__dlpack__`. On cupy>=13
# the toDlpack capsule is consumed on first use, causing a RuntimeError on the
# second call during backprop. Fix by swapping the branch order in thinc/util.py:
#
#   python - <<'EOF'
#   import site, os
#   util = os.path.join(site.getsitepackages()[0], "thinc", "util.py")
#   src = open(util).read()
#   old = (
#       '    if hasattr(xp_tensor, "toDlpack"):\n'
#       '        dlpack_tensor = xp_tensor.toDlpack()  # type: ignore\n'
#       '        torch_tensor = torch.utils.dlpack.from_dlpack(dlpack_tensor)\n'
#       '    elif hasattr(xp_tensor, "__dlpack__"):\n'
#       '        torch_tensor = torch.utils.dlpack.from_dlpack(xp_tensor)'
#   )
#   new = (
#       '    if hasattr(xp_tensor, "__dlpack__"):\n'
#       '        torch_tensor = torch.utils.dlpack.from_dlpack(xp_tensor)\n'
#       '    elif hasattr(xp_tensor, "toDlpack"):\n'
#       '        dlpack_tensor = xp_tensor.toDlpack()  # type: ignore\n'
#       '        torch_tensor = torch.utils.dlpack.from_dlpack(dlpack_tensor)'
#   )
#   open(util, "w").write(src.replace(old, new))
#   import glob; [os.remove(f) for f in glob.glob(os.path.dirname(util)+"/__pycache__/util*.pyc")]
#   print("Patched", util)
#   EOF
# ─────────────────────────────────────────────────────────────────────────────
